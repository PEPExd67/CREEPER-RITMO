// =========================================================================
// FIRMWARE CYBERPUNK DJ STUDIO ULTRA (VS Code + PlatformIO / Arduino IDE)
// Placa: Arduino Uno
// Hardware: 10 LEDs (Pines 12, 2, 3, 4, 5, 6, 7, 9, 10, 11) + Buzzer Pasivo (Pin 8)
// =========================================================================

#include <Arduino.h>

const int LEDS[10] = {12, 2, 3, 4, 5, 6, 7, 9, 10, 11};
const int TOTAL_LEDS = 10;
const int BUZZER_PIN = 8;

#define NOTE_C4  262
#define NOTE_E4  330
#define NOTE_G4  392
#define NOTE_C5  523
#define NOTE_E5  659
#define NOTE_G5  784
#define NOTE_C6  1047

void apagarTodos() {
  for (int i = 0; i < TOTAL_LEDS; i++) digitalWrite(LEDS[i], LOW);
}

void encenderTodos() {
  for (int i = 0; i < TOTAL_LEDS; i++) digitalWrite(LEDS[i], HIGH);
}

void sonarNota(int freq, int duracionMs) {
  if (freq > 0) {
    tone(BUZZER_PIN, freq, duracionMs);
    delay(duracionMs);
    noTone(BUZZER_PIN);
  } else {
    noTone(BUZZER_PIN);
    delay(duracionMs);
  }
}

void pad1Cowbell() {
  apagarTodos();
  digitalWrite(LEDS[7], HIGH); digitalWrite(LEDS[8], HIGH);
  tone(BUZZER_PIN, 587, 60); delay(65);
  tone(BUZZER_PIN, 845, 80); delay(85);
  noTone(BUZZER_PIN); apagarTodos();
}

void pad2SubDrop() {
  apagarTodos();
  digitalWrite(LEDS[0], HIGH); digitalWrite(LEDS[1], HIGH);
  for (int f = 220; f >= 30; f -= 20) { tone(BUZZER_PIN, f, 25); delay(25); }
  noTone(BUZZER_PIN); apagarTodos();
}

void pad3Explosion() {
  encenderTodos();
  for (int f = 900; f >= 80; f -= 100) { tone(BUZZER_PIN, f, 20); delay(20); }
  noTone(BUZZER_PIN); apagarTodos();
}

void pad4VocalChop() {
  apagarTodos();
  digitalWrite(LEDS[4], HIGH); digitalWrite(LEDS[5], HIGH);
  tone(BUZZER_PIN, 440, 40); delay(45);
  tone(BUZZER_PIN, 330, 70); delay(75);
  noTone(BUZZER_PIN); apagarTodos();
}

void pad5SlapBass() {
  apagarTodos();
  digitalWrite(LEDS[2], HIGH); digitalWrite(LEDS[3], HIGH);
  sonarNota(NOTE_E4, 80); apagarTodos();
}

void pad6Fuse() {
  for (int i = 0; i < 6; i++) {
    apagarTodos();
    digitalWrite(LEDS[i % TOTAL_LEDS], HIGH);
    digitalWrite(LEDS[(i + 5) % TOTAL_LEDS], HIGH);
    tone(BUZZER_PIN, random(1200, 2500), 25); delay(30);
  }
  noTone(BUZZER_PIN); apagarTodos();
}

void pad7Scratch() {
  for (int i = 0; i < 3; i++) {
    apagarTodos();
    digitalWrite(LEDS[i * 3], HIGH);
    tone(BUZZER_PIN, 800 + i * 200, 20); delay(25);
  }
  noTone(BUZZER_PIN); apagarTodos();
}

void pad8Laser() {
  for (int f = 1400; f >= 350; f -= 150) {
    apagarTodos();
    digitalWrite(LEDS[map(f, 350, 1400, 0, TOTAL_LEDS - 1)], HIGH);
    tone(BUZZER_PIN, f, 15); delay(18);
  }
  noTone(BUZZER_PIN); apagarTodos();
}

void pad9Glitch() {
  for (int i = 0; i < 5; i++) {
    apagarTodos();
    digitalWrite(LEDS[random(0, TOTAL_LEDS)], HIGH);
    tone(BUZZER_PIN, random(400, 1800), 20); delay(25);
  }
  noTone(BUZZER_PIN); apagarTodos();
}

void pad0Fanfare() {
  int notas[] = {NOTE_C5, NOTE_E5, NOTE_G5, NOTE_C6};
  for (int i = 0; i < 4; i++) {
    apagarTodos();
    for (int k = 0; k <= i * 2 + 1 && k < TOTAL_LEDS; k++) {
      digitalWrite(LEDS[k], HIGH);
    }
    sonarNota(notas[i], 70);
  }
  apagarTodos();
}

void setup() {
  Serial.begin(9600);
  for (int i = 0; i < TOTAL_LEDS; i++) pinMode(LEDS[i], OUTPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pad0Fanfare();
}

void loop() {
  if (Serial.available() > 0) {
    char cmd = Serial.read();
    switch (cmd) {
      case '1': case 'C': pad1Cowbell(); break;
      case '2': pad2SubDrop(); break;
      case '3': case 'X': pad3Explosion(); break;
      case '4': pad4VocalChop(); break;
      case '5': pad5SlapBass(); break;
      case '6': case 'F': pad6Fuse(); break;
      case '7': pad7Scratch(); break;
      case '8': pad8Laser(); break;
      case '9': pad9Glitch(); break;
      case '0': pad0Fanfare(); break;
      case 'D': pad3Explosion(); break;
    }
  }
}
